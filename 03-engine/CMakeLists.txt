cmake_minimum_required(VERSION 3.16)
project(hybrid_cpu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(HYBRID_NATIVE "Optimize for the build machine (binary may not be portable)" OFF)
find_package(Threads REQUIRED)
add_library(hybrid_runtime src/runtime.cpp)
target_include_directories(hybrid_runtime PUBLIC include)
target_link_libraries(hybrid_runtime PUBLIC Threads::Threads)
add_executable(hybrid-cpu src/main.cpp src/server.cpp)
target_link_libraries(hybrid-cpu PRIVATE hybrid_runtime)
if(WIN32)
  target_link_libraries(hybrid-cpu PRIVATE ws2_32 shell32)
endif()
if(MSVC)
  target_compile_options(hybrid_runtime PRIVATE /W4 /utf-8)
  target_compile_options(hybrid-cpu PRIVATE /W4 /utf-8)
  if(HYBRID_NATIVE)
    target_compile_options(hybrid_runtime PRIVATE /arch:AVX2)
  endif()
else()
  target_compile_options(hybrid_runtime PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(hybrid-cpu PRIVATE -Wall -Wextra -Wpedantic)
  if(HYBRID_NATIVE)
    target_compile_options(hybrid_runtime PRIVATE -march=native)
  endif()
endif()
